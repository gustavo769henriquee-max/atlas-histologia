const CAMPOS_MODIFICADOS = new Set();

function marcarModificado(chave) {
  if (!chave) return;

  CAMPOS_MODIFICADOS.add(chave);

  const input = document.querySelector(`[data-chave="${chave}"]`);

  console.debug(
    "[APARÃŠNCIA DEBUG] campo alterado:",
    JSON.stringify({ chave, valor: input ? input.value : "(sem input)" }),
  );
}

/* LÃª o objeto `tema` salvo com seguranÃ§a (fallback: objeto vazio) */
function obterTema(config = {}) {
  return config.tema && typeof config.tema === "object" ? config.tema : {};
}

/* Valor exibido num campo: configuraÃ§Ã£o salva â†’ senÃ£o padrÃ£o */
function valorTema(config, chave, padrao) {
  const tema = obterTema(config);

  return tema[chave] || padrao;
}

/* Abre um painel de aba respeitando a aba ativa salva (persistida) */
function abaAtivaSalva() {
  const salva = sessionStorage.getItem(CHAVE_ABA_APARENCIA);

  return ABAS_APARENCIA.some(([id]) => id === salva)
    ? salva
    : ABAS_APARENCIA[0][0];
}

function abrirAba(id) {
  const ativa = abaAtivaSalva();

  return `<div class="aparencia-aba" id="aba-${id}"${
    id === ativa ? "" : " hidden"
  }>`;
}

/* BotÃ£o de aba com estado (aria-selected + classe ativa) */
function estadoAba(id) {
  const ativa = abaAtivaSalva();

  return id === ativa
    ? ' class="aparencia-tab active" aria-selected="true"'
    : ' class="aparencia-tab" aria-selected="false"';
}

/* Coleta o tema atual do formulÃ¡rio (todas as cores + tipografia) */
function coletarTemaParaSalvar() {
  const tema = {};

  document.querySelectorAll("[data-chave]").forEach((input) => {
    const chave = input.dataset.chave;

    const valor = (input.value || "").trim();

    if (chave && corValida(valor)) tema[chave] = valor;
  });

  const fonte = document.querySelector("#config-fonte")?.value.trim();

  if (fonte) tema.fonte = fonte;

  const peso = document.querySelector("#config-peso-titulos")?.value;

  if (peso) tema.peso_titulos = peso;

  const tamanhoTitulo = document.querySelector("#config-tamanho-titulo")?.value;

  if (tamanhoTitulo) tema.tamanho_titulo = tamanhoTitulo;

  const tamanhoTexto = document.querySelector("#config-tamanho-texto")?.value;

  if (tamanhoTexto) tema.tamanho_texto = tamanhoTexto;

  return tema;
}

      const temaPronto = {};

      CAMPOS_MODIFICADOS.forEach((chave) => {

        if (temaColetado[chave] !== undefined && temaColetado[chave] !== "") {

      console.debug(
        "[APARÃŠNCIA DEBUG] campos modificados nesta submissÃ£o:",
        [...CAMPOS_MODIFICADOS],
      );

      console.debug("[APARÃŠNCIA DEBUG] temaPronto:", temaPronto);

      /*
       * Delta real desta submissÃ£o: somente chaves cujo valor difere do
       * efetivo atual. Se o usuÃ¡rio alterou uma cor, este objeto NÃƒO pode

      const deltaTema = temaParaSalvar(temaPronto, efetivoAtual);

      console.debug("[APARÃŠNCIA DEBUG] delta:", deltaTema);

      /*
       * Registro correto: SELECT e UPDATE passam a usar SEMPRE o mesmo id

      const { data: registroExistente } = await supabase

      let temaNoBanco = {};

        }
      }

      if (Object.keys(deltaTema).length > 0) {

      let colunaTemaRejeitada = false;

        resultado = idRegistro

        if (coluna === "tema") {

        resultado && !resultado.error && Array.isArray(resultado.data)

            `NÃ£o foi possÃ­vel salvar a aparÃªncia: ${resultado.error.message}`,

            "[APARÃŠNCIA] A gravaÃ§Ã£o nÃ£o afetou NENHUMA linha â€” polÃ­tica de RLS ou registro inexistente. Payload enviado:",
            dados,
          );

            "NÃ£o foi possÃ­vel salvar a aparÃªncia (permissÃ£o de gravaÃ§Ã£o negada).",
            "error",

        }

        falhaNoBotaoSalvar();

        return;
      }

      console.log("[APARÃŠNCIA] resposta do Supabase:", resultado.data);

      /*
       * CONFIRMAÃ‡ÃƒO REAL: reler o registro imediatamente e conferir se cada

          .maybeSingle();

      }

        verificacao &&
        verificacao.tema &&
        typeof verificacao.tema === "object" &&
        !Array.isArray(verificacao.tema)

      /*
       * Reaplica o tema no site real IMEDIATAMENTE (main.js escuta este
       * evento e relÃª configuracoes_site do banco â€” sem cache). Sem isto,
       * os tokens do documento sÃ³ eram atualizados na prÃ³xima navegaÃ§Ã£o.
       */
      window.dispatchEvent(new CustomEvent("atlas:config-atualizada"));

      setTimeout(() => {
        if (botaoSalvar) {
          atualizarRotuloBotaoSalvar("Salvar aparÃªncia");
        }
      }, 3000);
    } catch (erro) {
      console.error("[ATLAS] Erro inesperado ao salvar aparÃªncia:", erro);

      mostrarFeedbackSubmit(
        "NÃ£o foi possÃ­vel salvar a aparÃªncia.",
        "error",
      );

      if (botaoSalvar) {
        botaoSalvar.disabled = false;

        atualizarRotuloBotaoSalvar("Erro ao salvar");
      }

      setTimeout(() => {
        if (botaoSalvar) {
          atualizarRotuloBotaoSalvar("Salvar aparÃªncia");
        }
      }, 3000);
    } finally {
      form.dataset.salvando = "false";
    }
  });

  document
    .querySelector("#cancelar-aparencia")
    ?.addEventListener("click", () => {
      window.dispatchEvent(new Event("hashchange"));
    });
}

/*
 * Detecta nome de coluna inexistente em mensagens do PostgREST.
 * Formatos conhecidos:
 * - PGRST204: Could not find the 'tema' column of 'configuracoes_site' in the schema cache
 * - 42703:    column configuracoes_site.tema does not exist
 * Quando `candidatas` Ã© informado (payload atual), sÃ³ aceita nomes que
 * fazem parte dele â€” evita capturar identificadores alheios Ã  mensagem.
 */
function colunaDesconhecida(mensagem = "", candidatas = []) {
  const texto = String(mensagem);

  const citada = texto.match(/['"]([a-zA-Z_][a-zA-Z0-9_]*)['"]/);

  const pontuada = texto.match(/configuracoes_site\.([a-zA-Z_][a-zA-Z0-9_]*)/);

  const encontradas = [
    citada ? citada[1] : null,
    pontuada ? pontuada[1] : null,
  ].filter(Boolean);

  if (encontradas.length === 0) return null;

  if (candidatas.length > 0) {
    return encontradas.find((nome) => candidatas.includes(nome)) || null;
  }

  return encontradas[0];
}

/*
 * PRÃ‰VIA AO VIVO â€” usa os MESMOS tokens do site real.
 * Os valores sÃ£o aplicados como CSS custom properties no contÃªiner
 * da prÃ©via; como todo o site consome essas variÃ¡veis (inclusive os
 * aliases legados), a miniatura reflete fielmente qualquer mudanÃ§a.
 */
function configurarPreviewAoVivo() {
  const preview = document.querySelector("#aparencia-preview");

  if (!preview) return;

  /*
   * Coleta todos os campos (cores com data-chave + tipografia).
   * Valores invÃ¡lidos/vazios sÃ£o ignorados â€” quem decide o valor
   * efetivo Ã© temaEfetivo(), com fallback derivado seguro.
   */
  function coletarTema() {
    const tema = {};

    document.querySelectorAll("[data-chave]").forEach((input) => {
      const chave = input.dataset.chave;
      const valor = (input.value || "").trim();

      if (chave && corValida(valor)) tema[chave] = valor;
    });

    const fonte = document.querySelector("#config-fonte")?.value.trim();
    if (fonte) tema.fonte = fonte;

    const peso = document.querySelector("#config-peso-titulos")?.value;
    if (peso) tema.peso_titulos = peso;

    const tamanhoTitulo = document.querySelector("#config-tamanho-titulo")?.value;
    if (tamanhoTitulo) tema.tamanho_titulo = tamanhoTitulo;

    const tamanhoTexto = document.querySelector("#config-tamanho-texto")?.value;
    if (tamanhoTexto) tema.tamanho_texto = tamanhoTexto;

    return tema;
  }

  /* Aplica os MESMOS tokens do site real na prÃ©via */
  function aplicarAoVivo() {
    const base = temaEfetivo(temaSalvoAparencia);

    const delta = coletarTema();

    aplicarTema(preview, { ...base, ...delta });
  }

  /* Sincroniza seletor de cor <-> cÃ³digo hex e reaplica */
  function sincronizarCampo(campo) {
    const chave = campo.dataset.campoCor;

    const color = campo.querySelector('input[type="color"]');

    const codigo = campo.querySelector(".codigo-cor");

    if (!chave || !color || !codigo) return;

    color.addEventListener("input", () => {
      codigo.value = color.value.toUpperCase();

      marcarModificado(chave);

      aplicarAoVivo();
    });

    codigo.addEventListener("change", () => {
      let valor = codigo.value.trim();

      if (valor && !valor.startsWith("#")) valor = "#" + valor;

      if (corValida(valor)) {
        color.value = valor;

        marcarModificado(chave);

        aplicarAoVivo();
      } else {
        codigo.value = color.value.toUpperCase();
      }
    });
  }

  document.querySelectorAll(".campo-cor").forEach(sincronizarCampo);

  /* Textos da prÃ©via */
  function aplicarTextos() {
    const ler = (id, padrao) =>
      document.querySelector(id)?.value?.trim() || padrao;

    const definir = (seletor, id, padrao) => {

    definir(

    const marca = document.querySelector(".preview-footer-marca span");

    if (marca) {

  [

    "#config-botao-secundario",

  ].forEach((id) => {

  });

  [
    "#config-fonte",

  ].forEach((id) => {

    document.querySelector(id)?.addEventListener(evento, () => {
      /* Normaliza hÃ­fens: "peso-titulos" â†’ "peso_titulos" (chave do tema). */
      marcarModificado(id.replace("#config-", "").replace(/-/g, "_"));

}

function configurarTamanhoLogo() {
  const controle = document.querySelector("#config-logo-tamanho");

  if (!controle) return;

  const atualizar = () => {
    const tamanho = Number(controle.value || 100);

    const valorTexto = document.querySelector("#logo-tamanho-valor");

    if (valorTexto) {
      valorTexto.textContent = `${tamanho}%`;
    }

    const previewLogo = document.querySelector(".preview-logo");

    if (previewLogo) {
      const lado = Math.round(tamanho * 0.52);

      previewLogo.style.width = lado + "px";

      previewLogo.style.height = lado + "px";
    }

    const imagemEditor = document.querySelector("#logo-preview img");

    if (imagemEditor) {
      const ladoEditor = Math.round(90 * (tamanho / 100));

      imagemEditor.style.width = ladoEditor + "px";

      imagemEditor.style.height = ladoEditor + "px";
    }
  };

  controle.addEventListener("input", atualizar);

  atualizar();
}

function configurarLogoPreview() {
  const input = document.querySelector("#config-logo");

  const preview = document.querySelector("#logo-preview");

  const nome = document.querySelector("#logo-nome");

  const remover = document.querySelector("#remover-logo");

  if (!input || !preview || !nome || !remover) {
    return;
  }

  input.addEventListener("change", async () => {
    const arquivo = input.files[0];

    if (!arquivo) return;

    if (!arquivo.type.startsWith("image/")) {
      input.value = "";
      return;
    }

    nome.textContent = "Enviando logo...";

    const extensao = arquivo.name.split(".").pop().toLowerCase();

    const nomeArquivo = `logo-${Date.now()}.${extensao}`;

    const { error: uploadError } = await supabase.storage
      .from("logos")
      .upload(nomeArquivo, arquivo, {
        cacheControl: "3600",
        upsert: true,
      });

    if (uploadError) {
      console.error(uploadError);

      nome.textContent = "Erro ao enviar a logo";

      return;
    }

    const { data: urlData } = supabase.storage
      .from("logos")
      .getPublicUrl(nomeArquivo);

    const logoUrl = urlData.publicUrl;

    if (!logoUrl) {
      nome.textContent = "NÃ£o foi possÃ­vel obter a URL da logo";

      return;
    }

    logoUrlAtual = logoUrl;

    input.dataset.logoUrl = logoUrl;

    /* Mostra no editor */

    preview.innerHTML = "";

    const imagem = document.createElement("img");

    imagem.src = logoUrl;

    imagem.alt = "Logo do Atlas";

    preview.appendChild(imagem);

    nome.textContent = arquivo.name;

    /* Mostra tambÃ©m na prÃ©via do site */

    const previewLogo = document.querySelector(".preview-logo");

    if (previewLogo) {
      previewLogo.innerHTML = "";

      const imagemPreview = document.createElement("img");

      imagemPreview.src = logoUrl;

      imagemPreview.alt = "Logo";

      previewLogo.appendChild(imagemPreview);
    }

    /* Reaplica o tamanho escolhido */

    const controle = document.querySelector("#config-logo-tamanho");

    controle?.dispatchEvent(new Event("input"));
  });

  remover.addEventListener("click", () => {
    logoUrlAtual = "";

    delete input.dataset.logoUrl;

    input.value = "";

    preview.innerHTML = icone.microscopio;

    nome.textContent = "";

    const previewLogo = document.querySelector(".preview-logo");

    if (previewLogo) {
      previewLogo.innerHTML = icone.microscopio;
    }
  });
}

function escapeHtml(value = "") {
  return String(value)
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}
